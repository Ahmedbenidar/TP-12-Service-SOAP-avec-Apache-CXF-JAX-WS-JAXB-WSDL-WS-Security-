<soapenv:Envelope xmlns:soapenv="http://schemas.xmlsoap.org/soap/envelope/"
                  xmlns:ban="http://soap.ahmedbenidar.ma/banque/">

  <soapenv:Body>
    <ban:Saluer>
      <nomClient>Ahmed</nomClient>
    </ban:Saluer>

</soapenv:Envelope>
